import {Schema} from './integration/node_modules/effect/dist/index.js'
const IdentityFields={name:Schema.String.check(Schema.isPattern(/^movement-[0-9]+-[0-9]+\.json$/)),hash:Schema.String.check(Schema.isPattern(/^[a-f0-9]{64}$/))}
// Parent reads only this parsed artifact identity projection. Report values stay
// in the validator artifact, encoded through the production report codec.
export const ValidationManifestSchema=Schema.Struct({status:Schema.Literal('observed'),reports:Schema.Array(Schema.Struct(IdentityFields))})
export const makeValidationOutputSchema=reportSchema=>Schema.Struct({status:Schema.Literal('observed'),reports:Schema.Array(Schema.Struct({...IdentityFields,report:reportSchema}))})
