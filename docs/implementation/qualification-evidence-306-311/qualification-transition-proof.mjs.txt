export const verifySource=()=>{
 if(git('rev-parse','HEAD')!==successor||git('status','--porcelain'))throw Error('Clean successor required')
 const changed=git('diff','--name-only',oldCommit,successor).split('\n')
 if(changed.some(file=>!allow.includes(file)))throw Error('Unreviewed source transition')
 if(git('diff',oldCommit,successor,'--','src','packages/huly-cli/src'))throw Error('Production source changed')
 return hash(execFileSync('git',['diff','--binary',oldCommit,successor],{cwd:root}))
}

export const verifyHistorical=async()=>{
 const bytes=await readFile(summaryPath)
 if(hash(bytes)!=='f1f0d2dd5841bbddf53d32727660e1a12dda2e85e237f8648e8292797272f15d')throw Error('Historical summary changed')
 const summary=Schema.decodeUnknownSync(Schema.fromJsonString(SummarySchema))(bytes.toString('utf8')),parsed=summary.terminal
 const log=await readFile(parsed.log,'utf8')
 if(hash(log)!==parsed.logHash)throw Error('Historical log changed')
 const keys=log.split('\n').flatMap(line=>{const row=Schema.decodeUnknownOption(Schema.fromJsonString(AssertedRowSchema))(line);return row._tag==='Some'?[`${row.value.transport}:${row.value.case}`]:[]})
 if(JSON.stringify(keys)!==JSON.stringify(historical))throw Error('Historical asserted markers mismatch')
 const audit=await auditEvidenceArtifacts(parsed.evidenceDirectory,{files:parsed.artifactHashes,directories:parsed.artifactDirectories})
 const stage=parsed.movementStageEvidence
 if(!audit.filesStable||!audit.directoriesStable||!await auditReportArtifacts(stage.directory,stage.artifacts))throw Error('Historical artifacts changed')
 return hash(bytes)
}
export const reconstructGate=async gate=>{
 const projection=await mkdtemp('/tmp/hulymcp-dalph-306-311/takeover/transition-old-source-')
 try{
  const archive=execFileSync('git',['archive',oldCommit],{cwd:root,maxBuffer:64*1024*1024});execFileSync('tar',['-xf','-','-C',projection],{input:archive})
  await symlink(`${root}/node_modules`,`${projection}/node_modules`)
  await symlink(`${root}/dist`,`${projection}/dist`)
  await symlink(`${root}/packages/huly-cli/dist`,`${projection}/packages/huly-cli/dist`)
  const source=await readFile('/tmp/hulymcp-dalph-306-311/takeover/qualification-fingerprint.mjs','utf8')
  const helper=`${projection}/historical-fingerprint.mjs`
  await writeFile(helper,source.replaceAll("'./integration/scripts/run-movement-certification.mjs'","'./scripts/run-movement-certification.mjs'"))
  const original=await import(pathToFileURL(helper).href)
  const binding=await captureGateBinding(projection,original.fingerprintSuite)
  if(!gateBindingsEqual(binding,gate.binding))throw Error('Historical aggregate runtime binding not reproduced')
  return binding
 }finally{await rm(projection,{recursive:true,force:true})}
}
