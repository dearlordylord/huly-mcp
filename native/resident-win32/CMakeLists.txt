cmake_minimum_required(VERSION 3.24)
project(huly_resident_win32_security LANGUAGES CXX)

if(NOT WIN32)
  message(FATAL_ERROR "The resident security fixture requires a Windows target")
endif()

add_library(resident_win32_security STATIC security.cc)
target_compile_features(resident_win32_security PUBLIC cxx_std_17)
target_include_directories(resident_win32_security PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(resident_win32_security PUBLIC advapi32)

add_executable(resident_win32_security_test security-test.cc)
target_link_libraries(resident_win32_security_test PRIVATE resident_win32_security bcrypt)
foreach(target resident_win32_security resident_win32_security_test)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX /permissive- /analyze)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
  endif()
endforeach()
if(MINGW)
  target_link_options(resident_win32_security_test PRIVATE -municode)
endif()

enable_testing()
add_test(NAME resident_win32_security COMMAND resident_win32_security_test)
set_tests_properties(resident_win32_security PROPERTIES TIMEOUT 10)
